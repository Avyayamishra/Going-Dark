    86→    86→/**
    87→    87→ * Create (or return a cached) in-memory SQLite database for a story.
    88→    88→ */
    89→    89→async function getStoryDatabase(storyId: string): Promise<Database> {
    90→    90→  const cached = _dbCache.get(storyId);
    91→    91→  if (cached) return cached;
    92→    92→
    93→    93→  const story = STORY_REGISTRY.getStory(storyId);
    94→    94→  if (!story) {
    95→    95→    throw new Error(`Story not found: ${storyId}`);
    96→    96→  }
    97→    97→
    98→    98→  const SQL = await getSqlModule();
    99→    99→  const db = new SQL.Database();
   100→   100→  // Create the story's schema (each story defines its own tables).
   101→   101→  if (story.database.schema) {
   102→   102→    for (const tableDef of story.database.schema) {
   103→   103→      db.exec(tableDef.createSql);
   104→   104→    }
   105→   105→  } else {
   106→   106→    // Fallback to the shared schema (backward compatibility with Case 1).
   107→   107→    db.exec(SCHEMA_SQL);
   108→   108→  }
   109→   109→  // Seed all tables from the story's data.
   110→   110→  for (const table of story.database.tables) {
   111→   111→    seedTable(db, table);
   112→   112→  }
   113→   113→
   114→   114→  _dbCache.set(storyId, db);
   115→   115→  return db;
   116→   116→}

   131→   131→function seedTable(db: Database, table: SeedTable): void {
   132→   132→  if (table.rows.length === 0) return;
   133→   133→  const cols = Object.keys(table.rows[0]);
   134→   134→  const placeholders = cols.map(() => "?").join(",");
   135→   135→  const sql = `INSERT INTO ${table.name} (${cols.join(",")}) VALUES (${placeholders})`;
   136→   136→  const stmt = db.prepare(sql);
   137→   137→  for (const row of table.rows) {
   138→   138→    stmt.run(cols.map((c) => (row[c] ?? null) as SqlValue));
   139→   139→    stmt.reset();
   140→   140→  }
   141→   141→  stmt.free();
   142→   142→}

   251→   251→function stripComments(sql: string): string {
   252→   252→  let out = "";
   253→   253→  let i = 0;
   254→   254→  const n = sql.length;
   255→   255→  while (i < n) {
   256→   256→    const c = sql[i];
   257→   257→    const next = sql[i + 1];
   258→   258→    if (c === "-" && next === "-") {
   259→   259→      while (i < n && sql[i] !== "\n") i++;
   260→   260→      continue;
   261→   261→    }
   262→   262→    if (c === "/" && next === "*") {
   263→   263→      i += 2;
   264→   264→      while (i < n && !(sql[i] === "*" && sql[i + 1] === "/")) i++;
   265→   265→      i += 2;
   266→   266→      continue;
   267→   267→    }
   268→   268→    if (c === "'") {
   269→   269→      out += c;
   270→   270→      i++;
   271→   271→      while (i < n) {
   272→   272→        out += sql[i];
   273→   273→        if (sql[i] === "'") {
   274→   274→          if (sql[i + 1] === "'") {
   275→   275→            out += sql[i + 1];
   276→   276→            i += 2;
   277→   277→            continue;
   278→   278→          }
   279→   279→          i++;
   280→   280→          break;
   281→   281→        }
   282→   282→        i++;
   283→   283→      }
   284→   284→      continue;
   285→   285→    }
   286→   286→    if (c === '"') {
   287→   287→      out += c;
   288→   288→      i++;
   289→   289→      while (i < n) {
   290→   290→        out += sql[i];
   291→   291→        if (sql[i] === '"') {
   292→   292→          if (sql[i + 1] === '"') {
   293→   293→            out += sql[i + 1];
   294→   294→            i += 2;
   295→   295→            continue;
   296→   296→          }
   297→   297→          i++;
   298→   298→          break;
   299→   299→        }
   300→   300→        i++;
   301→   301→      }
   302→   302→      continue;
   303→   303→    }
   304→   304→    out += c;
   305→   305→    i++;
   306→   306→  }
   307→   307→  return out;
   308→   308→}

   310→   310→function splitStatements(sql: string): string[] {
   311→   311→  const stmts: string[] = [];
   312→   312→  let buf = "";
   313→   313→  let i = 0;
   314→   314→  const n = sql.length;
   315→   315→  while (i < n) {
   316→   316→    const c = sql[i];
   317→   317→    if (c === "'") {
   318→   318→      buf += c;
   319→   319→      i++;
   320→   320→      while (i < n) {
   321→   321→        buf += sql[i];
   322→   322→        if (sql[i] === "'") {
   323→   323→          if (sql[i + 1] === "'") {
   324→   324→            buf += sql[i + 1];
   325→   325→            i += 2;
   326→   326→            continue;
   327→   327→          }
   328→   328→          i++;
   329→   329→          break;
   330→   330→        }
   331→   331→        i++;
   332→   332→      }
   333→   333→      continue;
   334→   334→    }
   335→   335→    if (c === '"') {
   336→   336→      buf += c;
   337→   337→      i++;
   338→   338→      while (i < n) {
   339→   339→        buf += sql[i];
   340→   340→        if (sql[i] === '"') {
   341→   341→          if (sql[i + 1] === '"') {
   342→   342→            buf += sql[i + 1];
   343→   343→            i += 2;
   344→   344→            continue;
   345→   345→          }
   346→   346→          i++;
   347→   347→          break;
   348→   348→        }
   349→   349→        i++;
   350→   350→      }
   351→   351→      continue;
   352→   352→    }
   353→   353→    if (c === ";") {
   354→   354→      const s = buf.trim();
   355→   355→      if (s) stmts.push(s);
   356→   356→      buf = "";
   357→   357→      i++;
   358→   358→      continue;
   359→   359→    }
   360→   360→    buf += c;
   361→   361→    i++;
   362→   362→  }
   363→   363→  const last = buf.trim();
   364→   364→  if (last) stmts.push(last);
   365→   365→  return stmts;
   366→   366→}

   368→   368→const BLOCKED_KEYWORDS = [
   369→   369→  "INSERT", "UPDATE", "DELETE", "DROP", "ALTER", "CREATE", "ATTACH", "DETACH",
   370→   370→  "REPLACE", "REINDEX", "VACUUM", "ANALYZE", "PRAGMA", "BEGIN", "COMMIT",
   371→   371→  "ROLLBACK", "SAVEPOINT", "RELEASE", "LOAD", "IMPORT", "EXPORT", "EXPLAIN",
   372→   372→  "INTO", "VALUES",
   373→   373→];
   374→   374→
   375→   375→function findBlockedKeyword(tokenised: string[]): string | null {
   376→   376→  for (const kw of BLOCKED_KEYWORDS) {
   377→   377→    if (tokenised.includes(kw)) return kw;
   378→   378→  }
   379→   379→  return null;
   380→   380→}
   381→   381→
   382→   382→function tokenise(stmt: string): string[] {
   383→   383→  return stmt.toUpperCase().match(/[A-Z_]+/g) ?? [];
   384→   384→}
   385→   385→
   386→   386→function referencedTables(stmt: string): string[] {
   387→   387→  const tables: string[] = [];
   388→   388→  const upper = stmt.toUpperCase();
   389→   389→  const re = /(?:FROM|JOIN|INTO|UPDATE|TABLE)\s+([A-Za-z_][A-Za-z0-9_]*)/g;
   390→   390→  let m: RegExpExecArray | null;
   391→   391→  while ((m = re.exec(upper)) !== null) {
   392→   392→    tables.push(m[1].toLowerCase());
   393→   393→  }
   394→   394→  return tables;
   395→   395→}

   403→   403→export function validateQuery(rawSql: string, allowedTables?: Set<string>): ValidationResult {
   404→   404→  const cleaned = stripComments(rawSql ?? "").trim();
   405→   405→  if (!cleaned) {
   406→   406→    return {
   407→   407→      valid: false,
   408→   408→      statement: "",
   409→   409→      error: {
   410→   410→        kind: "EMPTY_QUERY",
   411→   411→        title: "EMPTY QUERY",
   412→   412→        message: "The query editor is empty. Write a SELECT statement and run it again.",
   413→   413→        hint: "Tip: try SELECT * FROM suspects;",
   414→   414→      },
   415→   415→    };
   416→   416→  }
   417→   417→
   418→   418→  const statements = splitStatements(cleaned);
   419→   419→  if (statements.length === 0) {
   420→   420→    return {
   421→   421→      valid: false,
   422→   422→      statement: "",
   423→   423→      error: {
   424→   424→        kind: "EMPTY_QUERY",
   425→   425→        title: "EMPTY QUERY",
   426→   426→        message: "The query editor is empty. Write a SELECT statement and run it again.",
   427→   427→      },
   428→   428→    };
   429→   429→  }
   430→   430→  if (statements.length > 1) {
   431→   431→    return {
   432→   432→      valid: false,
   433→   433→      statement: statements[0],
   434→   434→      error: {
   435→   435→        kind: "UNSUPPORTED",
   436→   436→        title: "MULTIPLE STATEMENTS",
   437→   437→        message: "Only a single read-only investigation query is permitted per execution.",
   438→   438→        hint: "Run one SELECT statement at a time.",
   439→   439→      },
   440→   440→    };
   441→   441→  }
   442→   442→
   443→   443→  const stmt = statements[0];
   444→   444→  const upper = stmt.toUpperCase();
   445→   445→  if (!/^(SELECT|WITH)\b/.test(upper)) {
   446→   446→    return {
   447→   447→      valid: false,
   448→   448→      statement: stmt,
   449→   449→      error: {
   450→   450→        kind: "UNSAFE_QUERY",
   451→   451→        title: "UNSAFE QUERY",
   452→   452→        message: "Only read-only investigation queries are permitted.",
   453→   453→        hint: "Queries must begin with SELECT or WITH.",
   454→   454→      },
   455→   455→    };
   456→   456→  }
   457→   457→
   458→   458→  const tokens = tokenise(stmt);
   459→   459→  const blocked = findBlockedKeyword(tokens);
   460→   460→  if (blocked) {
   461→   461→    return {
   462→   462→      valid: false,
   463→   463→      statement: stmt,
   464→   464→      error: {
   465→   465→        kind: "UNSAFE_QUERY",
   466→   466→        title: "UNSAFE QUERY",
   467→   467→        message: `Only read-only investigation queries are permitted. The keyword "${blocked}" is not allowed.`,
   468→   468→        hint: "Use SELECT to read from the investigation tables.",
   469→   469→      },
   470→   470→    };
   471→   471→  }
   472→   472→
   473→   473→  const tables = referencedTables(stmt);
   474→   474→  for (const t of tables) {
   475→   475→    if (!(allowedTables ?? DEFAULT_ALLOWED_TABLES).has(t)) {
   476→   476→      return {
   477→   477→        valid: false,
   478→   478→        statement: stmt,
   479→   479→        error: {
   480→   480→          kind: "NO_SUCH_TABLE",
   481→   481→          title: "QUERY ERROR",
   482→   482→          message: `Table "${t}" does not exist.`,
   483→   483→          hint: "Check the Database Explorer to see the available tables.",
   484→   484→        },
   485→   485→      };
   486→   486→    }
   487→   487→  }
   488→   488→
   489→   489→  return { valid: true, statement: stmt };
   490→   490→}

   492→   492→function classifyError(err: unknown): QueryFailure["error"] {
   493→   493→  const msg = err instanceof Error ? err.message : String(err);
   494→   494→  let m = msg.match(/no such table: (\S+)/i);
   495→   495→  if (m) {
   496→   496→    return {
   497→   497→      kind: "NO_SUCH_TABLE",
   498→   498→      title: "QUERY ERROR",
   499→   499→      message: `Table "${m[1]}" does not exist.`,
   500→   500→      hint: "Check the Database Explorer to see the available tables.",
   501→   501→    };
   502→   502→  }
   503→   503→  m = msg.match(/no such column: (\S+)/i);
   504→   504→  if (m) {
   505→   505→    return {
   506→   506→      kind: "NO_SUCH_COLUMN",
   507→   507→      title: "QUERY ERROR",
   508→   508→      message: `Column "${m[1]}" does not exist.`,
   509→   509→      hint: "Check the Database Explorer to see the available columns for each table.",
   510→   510→    };
   511→   511→  }
   512→   512→  if (/syntax error/i.test(msg) || /near ".*?": syntax error/i.test(msg)) {
   513→   513→    return {
   514→   514→      kind: "SYNTAX_ERROR",
   515→   515→      title: "QUERY ERROR",
   516→   516→      message: "The query contains a syntax error.",
   517→   517→      hint: "Check for missing commas, parentheses, or mismatched quotes. SQLite uses standard SQL syntax.",
   518→   518→    };
   519→   519→  }
   520→   520→  if (/ambiguous column name: (\S+)/i.test(msg)) {
   521→   521→    m = msg.match(/ambiguous column name: (\S+)/i);
   522→   522→    return {
   523→   523→      kind: "SYNTAX_ERROR",
   524→   524→      title: "QUERY ERROR",
   525→   525→      message: `Column "${m?.[1]}" is ambiguous — it exists in more than one table.`,
   526→   526→      hint: "Qualify the column with its table name, e.g. suspects.name",
   527→   527→    };
   528→   528→  }
   529→   529→  if (/no such function: (\S+)/i.test(msg)) {
   530→   530→    m = msg.match(/no such function: (\S+)/i);
   531→   531→    return {
   532→   532→      kind: "SYNTAX_ERROR",
   533→   533→      title: "QUERY ERROR",

   546→   546→const MAX_ROWS = 1000;
   547→   547→
   548→   548→export async function executeQuery(storyId: string, rawSql: string): Promise<QueryResult> {
   549→   549→  const start = performance.now();
   550→   550→  const validation = validateQuery(rawSql, getAllowedTables(storyId));
   551→   551→  if (!validation.valid) {
   552→   552→    return {
   553→   553→      ok: false,
   554→   554→      error: validation.error!,
   555→   555→      executionTimeMs: Math.round(performance.now() - start),
   556→   556→    };
   557→   557→  }
   558→   558→  const stmt = validation.statement;
   559→   559→  try {
   560→   560→    const db = await getStoryDatabase(storyId);
   561→   561→    let execResult: { columns: string[]; values: SqlValue[][] }[] = [];
   562→   562→    try {
   563→   563→      execResult = db.exec(stmt);
   564→   564→    } catch (err) {
   565→   565→      return {
   566→   566→        ok: false,
   567→   567→        error: classifyError(err),
   568→   568→        executionTimeMs: Math.round(performance.now() - start),
   569→   569→      };
   570→   570→    }
   571→   571→    let columns: string[] = [];
   572→   572→    let rows: QueryRow[] = [];
   573→   573→    let rowCount = 0;
   574→   574→    if (execResult.length > 0) {
   575→   575→      const r = execResult[0];
   576→   576→      columns = r.columns;
   577→   577→      rowCount = r.values.length;
   578→   578→      const limited = r.values.length > MAX_ROWS ? r.values.slice(0, MAX_ROWS) : r.values;
   579→   579→      rows = limited.map((rowVals) => {
   580→   580→        const obj: QueryRow = {};
   581→   581→        columns.forEach((c, idx) => {
   582→   582→          obj[c] = rowVals[idx];
   583→   583→        });
   584→   584→        return obj;
   585→   585→      });
   586→   586→    } else {
   587→   587→      columns = inferColumns(db, stmt);
   588→   588→      rowCount = 0;
   589→   589→      rows = [];
   590→   590→    }
   591→   591→    const truncated = rowCount > MAX_ROWS;
   592→   592→    return {
   593→   593→      ok: true,
   594→   594→      columns,
   595→   595→      rows,
   596→   596→      rowCount,
   597→   597→      executionTimeMs: Math.round(performance.now() - start),
   598→   598→      truncated,
   599→   599→    };
   600→   600→  } catch (err) {
   601→   601→    return {
   602→   602→      ok: false,
   603→   603→      error: classifyError(err),
   604→   604→      executionTimeMs: Math.round(performance.now() - start),
   605→   605→    };
   606→   606→  }
   607→   607→}

   609→   609→function inferColumns(db: Database, stmt: string): string[] {
   610→   610→  try {
   611→   611→    const inner = stmt.replace(/;\s*$/, "");
   612→   612→    const wrapped = `SELECT * FROM (${inner}) AS _q WHERE 1=0`;
   613→   613→    const res = db.exec(wrapped);
   614→   614→    if (res.length > 0) return res[0].columns;
   615→   615→    return [];
   616→   616→  } catch {
   617→   617→    return [];
   618→   618→  }
   619→   619→}

   752→   752→export async function getSchema(storyId: string): Promise<TableInfo[]> {
   753→   753→  const db = await getStoryDatabase(storyId);
   754→   754→  const story = STORY_REGISTRY.getStory(storyId);
   755→   755→  // If the story defines its own schema, use it; otherwise fall back to Case 1 tables.
   756→   756→  const tableNames = story?.database.schema?.map((s) => s.name) ?? ["suspects", "employees", "locations", "visits", "calls", "transactions", "messages", "security_logs"];
   757→   757→  return tableNames.map((t) => {
   758→   758→    const res = db.exec(`PRAGMA table_info(${t})`);
   759→   759→    const colsRaw: { name: string; type: string; notnull: number; pk: number }[] =
   760→   760→      res.length > 0
   761→   761→        ? res[0].values.map((v) => ({
   762→   762→            name: String(v[1]),
   763→   763→            type: String(v[2] || "TEXT"),
   764→   764→            notnull: Number(v[3]),
   765→   765→            pk: Number(v[5]),
   766→   766→          }))
   767→   767→        : [];
   768→   768→    // Use story-provided descriptions if available, otherwise fall back to defaults.
   769→   769→    const storyTableDef = story?.database.schema?.find((s) => s.name === t);
   770→   770→    const description = storyTableDef?.description ?? TABLE_DESCRIPTIONS[t] ?? "";
   771→   771→    return {
   772→   772→      name: t,
   773→   773→      description,
   774→   774→      columns: colsRaw.map((c) => {
   775→   775→        const storyColDef = storyTableDef?.columns.find((sc) => sc.name === c.name);
   776→   776→        return {
   777→   777→          name: c.name,
   778→   778→          type: c.type || "TEXT",
   779→   779→          notnull: c.notnull,
   780→   780→          pk: c.pk,
   781→   781→          description: storyColDef?.description ?? COLUMN_DESCRIPTIONS[t]?.[c.name] ?? "",
   782→   782→        };
   783→   783→      }),
   784→   784→    };
   785→   785→  });
   786→   786→}

   790→   790→export interface AccusationResult {
   791→   791→  correct: boolean;
   792→   792→  correctCount: number;
   793→   793→  who: { correct: boolean; answer: string; correctAnswer: string };

   797→   797→
   798→   798→export function evaluateAccusation(
   799→   799→  storyId: string,
   800→   800→  who: string,
   801→   801→  how: string,
   802→   802→  why: string,
   803→   803→): AccusationResult {
   804→   804→  const story = STORY_REGISTRY.getStory(storyId);
   805→   805→  if (!story) {
   806→   806→    return {
   807→   807→      correct: false,
   808→   808→      correctCount: 0,
   809→   809→      who: { correct: false, answer: who, correctAnswer: "" },
   810→   810→      how: { correct: false, answer: how, correctAnswer: "" },
   811→   811→      why: { correct: false, answer: why, correctAnswer: "" },
   812→   812→    };
   813→   813→  }
   814→   814→  const CORRECT = story.solution;
   815→   815→  const whoCorrect = who === CORRECT.who;
   816→   816→  const howCorrect = how === CORRECT.how;
   817→   817→  const whyCorrect = why === CORRECT.why;
   818→   818→  return {
   819→   819→    correct: whoCorrect && howCorrect && whyCorrect,
   820→   820→    correctCount: [whoCorrect, howCorrect, whyCorrect].filter(Boolean).length,
   821→   821→    who: { correct: whoCorrect, answer: who, correctAnswer: CORRECT.who },
   822→   822→    how: { correct: howCorrect, answer: how, correctAnswer: CORRECT.how },
   823→   823→    why: { correct: whyCorrect, answer: why, correctAnswer: CORRECT.why },
   824→   824→  };
   825→   825→}

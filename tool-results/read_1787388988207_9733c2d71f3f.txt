   149→// ---------- DATA ----------
   150→// Helper: bulk-insert rows into a table using positional ? placeholders.
   151→function insertRows(table: string, rows: Record<string, unknown>[]) {
   152→  if (rows.length === 0) return;
   153→  const cols = Object.keys(rows[0]);
   154→  const placeholders = cols.map(() => "?").join(",");
   155→  const sql = `INSERT INTO ${table} (${cols.join(",")}) VALUES (${placeholders})`;
   156→  const stmt = db.prepare(sql);
   157→  for (const row of rows) {
   158→    stmt.run(cols.map((c) => row[c] ?? null));
   159→    stmt.reset();
   160→  }
   161→  stmt.free();
   162→}
